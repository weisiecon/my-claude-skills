% ----- Custom colours -----
\definecolor{commentbg}{gray}{0.93}
\definecolor{commentborder}{gray}{0.75}

% ----- Referee comment environment -----
% Sans-serif text on grey background
\newenvironment{refcomment}{%
  \begin{mdframed}[
    backgroundcolor=commentbg,
    linecolor=commentborder,
    linewidth=0.5pt,
    innertopmargin=8pt,
    innerbottommargin=8pt,
    innerleftmargin=10pt,
    innerrightmargin=10pt,
    skipabove=10pt,
    skipbelow=6pt
  ]%
  \sffamily\small%
}{%
  \end{mdframed}%
}

% ----- Response environment -----
% Serif text, normal formatting
\newenvironment{response}{%
  \par\noindent\textbf{Response:}\quad\ignorespaces%
}{%
  \par\vspace{6pt}%
}

% ----- Manuscript reference command -----
\newcommand{\manuscriptref}[2]{%
  \par\noindent\textit{See Section~#1, page~#2 of the revised manuscript.}%
}

% ----- Reviewer section command -----
\newcommand{\reviewersection}[1]{%
  \clearpage%
  \section*{Response to Reviewer #1}%
  \addcontentsline{toc}{section}{Response to Reviewer #1}%
}

% ----- Comment heading command -----
\newcommand{\commentheading}[1]{%
  \subsection*{Comment #1}%
}
